4 * Opt-in JSON result sink. Buffers result rows and emits a single valid JSON
5 * array (one object per result) on flush() / destruction. The `name` string is
6 * escaped per JSON rules. Header-only and dependency-free — include
7 * <bench/sink/json.hpp> explicitly (NOT in <bench/all>).
8 */
9#ifndef BENCH_SINK_JSON_HPP
10#define BENCH_SINK_JSON_HPP
11
12#include <ostream>
13#include <vector>
14#include "../bench.hpp"
15
16namespace bench {
17
18 // Buffers results, then writes one JSON array on flush() or at destruction.
19 // The caller owns the stream's lifetime; this sink only borrows it.
20 struct json_sink : sink {
21 explicit json_sink(std::ostream& os) : os_(os) {}
22 ~json_sink() override { flush(); }
23
24 void write(const result_t& r) override { rows_.push_back(r); }
25
26 void flush() override {
27 if(emitted_) return;
28 emitted_ = true;
29 os_ << '[';
30 for(std::size_t i = 0; i < rows_.size(); ++i){
31 const result_t& r = rows_[i];
32 if(i) os_ << ',';
33 os_ << "{\"name\":\"";
34 escape(r.name);
35 os_ << "\",\"warmup\":" << r.warmup
36 << ",\"sample\":" << r.sample
37 << ",\"x\":" << r.x
38 << ",\"mean_runtime\":" << r.mean_runtime
39 << ",\"std_runtime\":" << r.std_runtime
40 << ",\"mean_throughput\":" << r.mean_throughput
41 << ",\"std_throughput\":" << r.std_throughput
42 << '}';
43 }
44 os_ << ']';
45 os_.flush();
46 }
47
48 private:
49 void escape(const char* s){
50 for(; *s; ++s){
51 const unsigned char c = static_cast<unsigned char>(*s);
52 switch(c){
53 case '"': os_ << "\\\""; break;
54 case '\\': os_ << "\\\\"; break;
55 case '\b': os_ << "\\b"; break;
56 case '\f': os_ << "\\f"; break;
57 case '\n': os_ << "\\n"; break;
58 case '\r': os_ << "\\r"; break;
59 case '\t': os_ << "\\t"; break;
60 default:
61 if(c < 0x20){
62 static const char hex[] = "0123456789abcdef";
63 os_ << "\\u00" << hex[(c >> 4) & 0xF] << hex[c & 0xF];
64 } else {
65 os_ << static_cast<char>(c);
66 }
67 }
68 }
69 }
70 std::ostream& os_;
71 std::vector<result_t> rows_;
72 bool emitted_ = false;
73 };
74}
75
76#endif
