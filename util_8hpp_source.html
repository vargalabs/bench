3 *
4 * Test-data generators. Header-only, dependency-free (standard library only).
5 *
6 * The single entry point is `bench::util::get_test_data<T>(count, seed, width)`
7 * which returns a deterministic `std::vector<T>` of generated test data. Given
8 * the same (count, seed, width) it produces byte-identical output on every run,
9 * so benchmark inputs are fully reproducible.
10 *
11 * - arithmetic T (int/float/double/...): seeded std::mt19937; `width` ignored.
12 * - std::string : `width` is the max/target length of
13 * each generated printable-ASCII string.
14 */
15#ifndef BENCH_UTIL_HPP
16#define BENCH_UTIL_HPP
17
18#include <cstddef>
19#include <limits>
20#include <random>
21#include <string>
22#include <type_traits>
23#include <vector>
24
25namespace bench::util {
26
27 namespace detail {
28 // Printable ASCII range used for generated strings: '!' (33) .. '~' (126).
29 inline constexpr char ascii_lo = '!';
30 inline constexpr char ascii_hi = '~';
31 inline constexpr std::size_t default_string_width = 16;
32 } // namespace detail
33
34 // Primary template handles arithmetic T. std::string is split off below via
35 // `if constexpr`, keeping a single user-facing signature.
36 template<class T>
37 std::vector<T> get_test_data(std::size_t count, unsigned seed = 0, std::size_t width = 0) {
38 static_assert(std::is_arithmetic_v<T> || std::is_same_v<T, std::string>,
39 "bench::util::get_test_data supports arithmetic types and std::string");
40
41 std::vector<T> out;
42 out.reserve(count);
43 std::mt19937 gen(seed);
44
45 if constexpr (std::is_same_v<T, std::string>) {
46 // width == 0 -> use a sensible default target length.
47 const std::size_t target = (width == 0) ? detail::default_string_width : width;
48 // Each string has a length in [1, target]; chars are printable ASCII.
49 std::uniform_int_distribution<std::size_t> len_dist(1, target);
50 std::uniform_int_distribution<int> chr_dist(detail::ascii_lo, detail::ascii_hi);
51 for (std::size_t i = 0; i < count; ++i) {
52 const std::size_t n = len_dist(gen);
53 std::string s;
54 s.reserve(n);
55 for (std::size_t j = 0; j < n; ++j)
56 s.push_back(static_cast<char>(chr_dist(gen)));
57 out.push_back(std::move(s));
58 }
59 } else if constexpr (std::is_floating_point_v<T>) {
60 std::uniform_real_distribution<T> dist(T(0), T(1));
61 for (std::size_t i = 0; i < count; ++i)
62 out.push_back(dist(gen));
63 } else { // integral
64 using limits = std::numeric_limits<T>;
65 // std::uniform_int_distribution is only defined for short/int/long/
66 // long long and their unsigned variants — NOT char/bool/int8_t/
67 // uint8_t/char8_t. libstdc++ permits the small types as an extension,
68 // but MSVC enforces the standard (static_assert). Generate through a
69 // wide conforming type and narrow to T.
70 using dist_t = std::conditional_t<std::is_signed_v<T>, long long, unsigned long long>;
71 std::uniform_int_distribution<dist_t> dist(
72 static_cast<dist_t>(limits::min()), static_cast<dist_t>(limits::max()));
73 for (std::size_t i = 0; i < count; ++i)
74 out.push_back(static_cast<T>(dist(gen)));
75 }
76 return out;
77 }
78
79} // namespace bench::util
