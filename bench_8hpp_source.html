45 template <typename Tag, int N>
46 struct array_t {
47 using value_type = Tag;
48 array_t(const std::initializer_list<std::size_t>& list ) : rank(0){
49 for(std::size_t v: list) value[rank] = v, rank++;
50 }
51 array_t() : rank(0){};
52 array_t( array_t&& arg ) = default;
53 array_t( const array_t& arg ) = default;
54 array_t& operator=( array_t&& arg ) = default;
55 array_t& operator=( const array_t& arg ) = default;
56
57 std::size_t& operator[](std::size_t i){ return *(value + i); }
58 const std::size_t& operator[](std::size_t i) const { return *(value + i); }
59 std::size_t* operator*() { return value; }
60 const std::size_t* operator*() const { return value; }
61
62 std::size_t size() const { return rank; }
63 const std::size_t* begin()const { return value; }
64 std::size_t* begin() { return value; }
65 const std::size_t* end() const { return value+rank; }
66 std::size_t* end() { return value+rank; }
67
68 std::size_t value[N];
69 std::size_t rank;
70 };

72 // Dependency-free, RTTI-free compile-time type name. Extracts the spelled-out
73 // type from the compiler's pretty-function macro; falls back to an empty label
74 // on unknown compilers (the caller then suffixes a numeric index instead).
75 template <class T>
76 constexpr std::string type_name() {
77#if defined(__clang__)
78 // "std::string bench::impl::type_name() [T = int]"
79 constexpr const char* p = __PRETTY_FUNCTION__;
80 std::string_view sv(p);
81 const auto b = sv.find("[T = ");
82 if (b == std::string_view::npos) return std::string();
83 const auto s = b + 5;
84 const auto e = sv.rfind(']');
85 return std::string(sv.substr(s, e - s));
86#elif defined(__GNUC__)
87 // "constexpr std::string bench::impl::type_name() [with T = int; ...]"
88 constexpr const char* p = __PRETTY_FUNCTION__;
89 std::string_view sv(p);
90 const auto b = sv.find("[with T = ");
91 if (b == std::string_view::npos) return std::string();
92 const auto s = b + 10;
93 auto e = sv.find(';', s);
94 if (e == std::string_view::npos) e = sv.rfind(']');
95 return std::string(sv.substr(s, e - s));
96#elif defined(_MSC_VER)
97 // "...type_name<int>(void)"
98 constexpr const char* p = __FUNCSIG__;
99 std::string_view sv(p);
100 const auto b = sv.find("type_name<");
101 if (b == std::string_view::npos) return std::string();
102 const auto s = b + 10;
103 const auto e = sv.rfind('>');
104 return std::string(sv.substr(s, e - s));
105#else
106 return std::string();
107#endif
108 }

110 // mean / population-stddev. FIX vs the original POC: the accumulate seed must
111 // be T{0} (a double), not the integer literal 0, which truncated the result.
112 template <class T>
113 std::pair<T,T> stats(const std::vector<T>& data){
114 if (data.empty()) return {T{0}, T{0}};
115 T mean = std::accumulate(data.begin(), data.end(), T{0}) / static_cast<T>(data.size());
116 T var = std::accumulate(data.begin(), data.end(), T{0}, [&](const T& s, const T& v ) -> T {
117 T m = (v-mean);
118 return s + m*m;
119 }) / static_cast<T>(data.size());
120 return std::pair<T,T>{mean, std::sqrt(var)};
121 }

123 // Detects whether `T` is a specialization of the class template `Tmpl`.
124 // Used to constrain the type-axis throughput overload to `std::tuple<...>`.
125 template <class T, template <class...> class Tmpl>
126 struct is_specialization_of : std::false_type {};
127 template <template <class...> class Tmpl, class... Us>
128 struct is_specialization_of<Tmpl<Us...>, Tmpl> : std::true_type {};
129
130 template <class T>
131 concept is_tuple = is_specialization_of<std::remove_cvref_t<T>, std::tuple>::value;
132}

152 // ---- result record (plain POD; was the HDF5-registered test_t) ---------
153 struct result_t {
154 static constexpr std::size_t max_name = 64;
155 std::uint16_t warmup;
156 std::uint16_t sample;
157 std::uint64_t x;
158 double mean_runtime; // microseconds
159 double std_runtime; // microseconds
160 double mean_throughput; // bytes / microsecond (== MB/s)
161 double std_throughput;
162 char name[max_name];
163 };

165 // ---- pluggable result sink --------------------------------------------
166 struct sink {
167 virtual ~sink() = default;
168 virtual void write(const result_t&) = 0;
169 virtual void flush() {}
170 };
171
172 // default sink: the fixed-width table the POC printed (now dependency-free).
173 struct stdout_sink : sink {
174 bool header = false;
175 void write(const result_t& r) override {
176 if(!header){
177 std::printf("[name ]"
178 "[total events][Mi events/s] [ms runtime / stddev] [ MiB/s / stddev ]\n");
179 header = true;
180 }
181 std::printf("%-52s %12llu %12.4f %11.2f %8.3f %10.2f %7.1f\n",
182 r.name, static_cast<unsigned long long>(r.x),
183 r.mean_runtime > 0 ? r.x / r.mean_runtime : 0.0,
184 r.mean_runtime/1000.0, r.std_runtime/1000.0,
185 r.mean_throughput, r.std_throughput);
186 }
187 };
188
189 // ---- in-memory store; emits to the sink at program exit ----------------
190 struct store_t {
191 store_t(store_t const&) = delete;
192 void operator=(store_t const&) = delete;
193 static store_t& get() { static store_t instance; return instance; }
194
195 void set_sink(std::shared_ptr<sink> s){ sink_ = std::move(s); }
196 void push(const result_t& data){ list_.push_back(data); }
197
198 template<class V, class T>
199 void push(const std::string& nm, std::uint64_t x, V wu, V su,
200 const std::pair<T,T>& time, const std::pair<T,T>& tp ) {
201 result_t item{};
202 item.warmup = wu;
203 item.sample = su;
204 item.x = x;
205 item.mean_runtime = time.first;
206 item.std_runtime = time.second;
207 item.mean_throughput = tp.first;
208 item.std_throughput = tp.second;
209 std::strncpy(item.name, nm.data(), result_t::max_name - 1);
210 list_.push_back(item);
211 }
212
213 const std::vector<result_t>& results() const { return list_; }
214
215 private:
216 store_t() : sink_(std::make_shared<stdout_sink>()) {}
217 ~store_t(){
218 for(const auto& r: list_) sink_->write(r);
219 sink_->flush();
220 }
221 std::shared_ptr<sink> sink_;
222 std::vector<result_t> list_;
223 };
224
225 inline void set_sink(std::shared_ptr<sink> s){ store_t::get().set_sink(std::move(s)); }

227 // ---- shared per-axis timing loop --------------------------------------
228 // `fn(j, n)` returns bytes moved; `before`/`after` are the (optional) sample
229 // hooks. Pushes one result_t row per x-axis point. Single source of truth so
230 // the scalar and type-axis drivers time identically.
231 namespace impl {
232 template <class fn_t, class before_t, class after_t>
233 void run_axis(store_t& store, const std::string& nm,
234 const arg_x& x, std::uint16_t wu, std::uint16_t su,
235 fn_t&& fn, before_t&& before, after_t&& after) {
236 std::vector<double> time, tput;
237 for(std::size_t j=0; j<x.size(); j++){
238 for(std::uint16_t i=0; i<wu; i++){ volatile double s = fn(j, x[j]); (void)s; }
239 time.clear(); tput.clear();
240 for(std::uint16_t i=0; i<su; i++) {
241 namespace cl = std::chrono;
242 before();
243 const auto start = cl::steady_clock::now();
244 const double transfer = fn(j, x[j]);
245 after();
246 const auto end = cl::steady_clock::now();
247 const double delta_us = static_cast<double>(
248 cl::duration_cast<cl::nanoseconds>(end - start).count()) / 1'000.0;
249 time.push_back(delta_us);
250 tput.push_back(delta_us > 0 ? transfer / delta_us : 0.0);
251 }
252 store.push(nm, x[j], wu, su, bench::impl::stats(time), bench::impl::stats(tput));
253 }
254 }
255 }

257 // ---- the throughput driver --------------------------------------------
258 // body signature: double(std::size_t idx, std::size_t n) -> bytes moved.
259 // The compile-time type-axis form is the explicit-tuple overload below
260 // (`throughput<std::tuple<Ts...>>(...)`).
261 template <class... args_t>
262 void throughput(args_t... args) {
263 using callback_t = std::function<double(std::size_t, std::size_t)>;
264
265 using name_t = typename arg::tpos<impl::tag::name_t, args_t...>;
266 using x_t = typename arg::tpos<impl::tag::x_t, args_t...>;
267 using warmup_t = typename arg::tpos<impl::tag::warmup_t, args_t...>;
268 using sample_t = typename arg::tpos<impl::tag::sample_t, args_t...>;
269 using before_t = typename arg::tpos<impl::tag::before_sample_t, args_t...>;
270 using after_t = typename arg::tpos<impl::tag::after_sample_t, args_t...>;
271 using callback_f = typename arg::tpos<callback_t, args_t...>;
272
273 static_assert( name_t::present, "benchmark name must be specified (bench::name{...})" );
274 static_assert( x_t::present, "x axis must be specified (bench::arg_x{...})" );
275 static_assert( callback_f::present, "benchmark body callback must be specified" );
276
277 auto tuple = std::forward_as_tuple(args...);
278 store_t& store = store_t::get();
279
280 name nm = std::get<name_t::position>(tuple);
281 arg_x x = std::get<x_t::position>(tuple);
282 std::uint16_t wu = 3, su = 20;
283 if constexpr(warmup_t::present) wu = std::get<warmup_t::position>(tuple).value;
284 if constexpr(sample_t::present) su = std::get<sample_t::position>(tuple).value;
285
286 callback_t fn = std::get<callback_f::position>(tuple);
287
288 const auto before = [&]{ if constexpr(before_t::present) std::get<before_t::position>(tuple).value(); };
289 const auto after = [&]{ if constexpr(after_t::present) std::get<after_t::position>(tuple).value(); };
290
291 impl::run_axis(store, nm.value, x, wu, su, fn, before, after);
292 }

294 // ---- the type-axis throughput driver ----------------------------------
295 // Folds the benchmark body over the compile-time typelist supplied as an
296 // explicit template argument, a `std::tuple<Ts...>`. Invoke as
297 // bench::throughput<std::tuple<A,B,C>>(bench::name{...}, ...);
298 // The leading template parameter `Tuple` is constrained (impl::is_tuple) to
299 // be a `std::tuple` specialization so this never collides with the scalar
300 // `throughput(args...)` overload — ordinary calls deduce nothing for `Tuple`
301 // and select the scalar form, while an explicit tuple selects this one.
302 //
303 // `body` is a C++20 generic lambda invoked once per type as
304 // body.template operator()<T>(std::size_t idx, std::size_t n) -> double
305 // returning bytes moved (same contract as the scalar driver). Honours the
306 // same named args/hooks; one result row per (type, x-point), the row name
307 // suffixed with the type label so types are distinguishable.
308 template <class Tuple, class... args_t>
309 requires impl::is_tuple<Tuple>
310 void throughput(args_t... args) {
311 using name_t = typename arg::tpos<impl::tag::name_t, args_t...>;
312 using x_t = typename arg::tpos<impl::tag::x_t, args_t...>;
313 using warmup_t = typename arg::tpos<impl::tag::warmup_t, args_t...>;
314 using sample_t = typename arg::tpos<impl::tag::sample_t, args_t...>;
315 using before_t = typename arg::tpos<impl::tag::before_sample_t, args_t...>;
316 using after_t = typename arg::tpos<impl::tag::after_sample_t, args_t...>;
317
318 static_assert( name_t::present, "benchmark name must be specified (bench::name{...})" );
319 static_assert( x_t::present, "x axis must be specified (bench::arg_x{...})" );
320 static_assert( std::tuple_size_v<Tuple> > 0, "type axis must list at least one type" );
321
322 auto tuple = std::forward_as_tuple(args...);
323 store_t& store = store_t::get();
324
325 name nm = std::get<name_t::position>(tuple);
326 arg_x x = std::get<x_t::position>(tuple);
327 std::uint16_t wu = 3, su = 20;
328 if constexpr(warmup_t::present) wu = std::get<warmup_t::position>(tuple).value;
329 if constexpr(sample_t::present) su = std::get<sample_t::position>(tuple).value;
330
331 const auto before = [&]{ if constexpr(before_t::present) std::get<before_t::position>(tuple).value(); };
332 const auto after = [&]{ if constexpr(after_t::present) std::get<after_t::position>(tuple).value(); };
333
334 // the generic body is the only positional (un-tagged) argument
335 auto&& body = arg::getn<sizeof...(args_t) - 1>(args...);
336
337 meta::impl::static_for<Tuple>([&](auto I){
338 using T = std::tuple_element_t<decltype(I)::value, Tuple>;
339 std::string label = impl::type_name<T>();
340 if (label.empty()) label = "T" + std::to_string(decltype(I)::value);
341 const std::string row_name = nm.value + "/" + label;
342
343 impl::run_axis(store, row_name, x, wu, su,
344 [&](std::size_t idx, std::size_t n) -> double {
345 return body.template operator()<T>(idx, n);
346 }, before, after);
347 });
348 }
